你想要一个能在 Windows 系统上运行的简易拍照软件，核心需求是调用电脑摄像头完成拍照、预览，并能将照片保存到本地，对吧？

我会用 Python 来实现这个软件，它跨平台且对新手友好，代码简洁易懂，还能打包成 Windows 可执行文件（.exe），不用装 Python 也能直接用。

### 实现思路
1. 使用 `opencv-python` 调用电脑摄像头，实现实时预览
2. 使用 `tkinter` 搭建简单的图形界面（按钮、预览窗口）
3. 点击按钮触发拍照动作，将照片以时间戳命名并保存到指定文件夹
4. 支持退出摄像头、关闭软件等基础操作

### 完整代码
```python
import cv2
import tkinter as tk
from tkinter import messagebox
from PIL import Image, ImageTk
import datetime
import os

# 确保保存照片的文件夹存在
SAVE_DIR = "CameraPhotos"
if not os.path.exists(SAVE_DIR):
    os.makedirs(SAVE_DIR)

class WindowsCameraApp:
    def __init__(self, window):
        self.window = window
        self.window.title("Windows 简易拍照软件")
        self.window.geometry("800x650")  # 窗口大小
        self.window.resizable(False, False)  # 固定窗口大小

        # 初始化摄像头
        self.cap = cv2.VideoCapture(0)  # 0 表示默认摄像头，多个摄像头可换 1、2 等
        if not self.cap.isOpened():
            messagebox.showerror("错误", "无法打开摄像头！请检查摄像头是否被占用或驱动正常。")
            self.window.quit()
            return

        # 创建预览标签
        self.label = tk.Label(window)
        self.label.pack(pady=10)

        # 创建按钮框架
        btn_frame = tk.Frame(window)
        btn_frame.pack(pady=10)

        # 拍照按钮
        self.capture_btn = tk.Button(
            btn_frame, text="拍照", font=("微软雅黑", 12),
            width=10, command=self.take_photo
        )
        self.capture_btn.grid(row=0, column=0, padx=10)

        # 退出按钮
        self.quit_btn = tk.Button(
            btn_frame, text="退出", font=("微软雅黑", 12),
            width=10, command=self.close_app
        )
        self.quit_btn.grid(row=0, column=1, padx=10)

        # 显示保存路径提示
        tip_label = tk.Label(
            window, text=f"照片将保存到：{os.path.abspath(SAVE_DIR)}",
            font=("微软雅黑", 10), fg="gray"
        )
        tip_label.pack(pady=5)

        # 实时更新摄像头画面
        self.update_frame()

    def update_frame(self):
        """更新摄像头预览画面"""
        ret, frame = self.cap.read()
        if ret:
            # 转换颜色空间（OpenCV 是 BGR，PIL 是 RGB）
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            # 调整画面大小适配窗口
            frame = cv2.resize(frame, (760, 480))
            # 转换为 PIL 图像格式
            img = Image.fromarray(frame)
            imgtk = ImageTk.PhotoImage(image=img)
            self.label.imgtk = imgtk
            self.label.config(image=imgtk)
        # 循环更新画面（每10毫秒刷新一次）
        self.window.after(10, self.update_frame)

    def take_photo(self):
        """拍照并保存照片"""
        ret, frame = self.cap.read()
        if ret:
            # 生成带时间戳的文件名，避免重复
            timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
            photo_path = os.path.join(SAVE_DIR, f"photo_{timestamp}.jpg")
            # 保存照片
            cv2.imwrite(photo_path, frame)
            messagebox.showinfo("成功", f"照片已保存！\n路径：{photo_path}")
        else:
            messagebox.showerror("错误", "拍照失败，请重试！")

    def close_app(self):
        """关闭软件，释放摄像头"""
        self.cap.release()  # 释放摄像头资源
        self.window.quit()  # 关闭窗口

if __name__ == "__main__":
    root = tk.Tk()
    app = WindowsCameraApp(root)
    root.mainloop()
```

### 使用说明
#### 1. 环境准备
首先需要安装依赖库，打开 Windows 命令提示符（CMD），执行以下命令：
```bash
pip install opencv-python pillow
```
- `opencv-python`：用于调用摄像头、处理图像
- `pillow`：用于将 OpenCV 图像转换为 TKinter 可显示的格式

#### 2. 运行代码
- 将代码保存为 `camera_app.py`
- 双击运行（或在 CMD 中执行 `python camera_app.py`）
- 软件会自动打开默认摄像头，显示实时预览
- 点击「拍照」按钮，照片会保存到当前目录下的 `CameraPhotos` 文件夹，文件名带时间戳
- 点击「退出」按钮，会释放摄像头并关闭软件

#### 3. 打包成 EXE（可选，方便无 Python 环境使用）
如果想做成无需安装 Python 就能运行的 EXE 文件，需要安装 `pyinstaller`：
```bash
pip install pyinstaller
```
然后执行打包命令：
```bash
pyinstaller -F -w -i camera.ico camera_app.py
```
- `-F`：打包成单个 EXE 文件
- `-w`：无控制台窗口（纯图形界面）
- `-i camera.ico`：可选，设置软件图标（需提前准备 ico 格式图标）
- 打包完成后，EXE 文件会出现在 `dist` 文件夹中

### 注意事项
1. 运行前确保摄像头未被其他软件（如微信、钉钉、Zoom）占用
2. 如果有多个摄像头，可将代码中 `cv2.VideoCapture(0)` 的 `0` 改为 `1`/`2` 切换
3. 保存的照片默认在软件运行目录的 `CameraPhotos` 文件夹，可修改代码中 `SAVE_DIR` 变量自定义路径

### 总结
1. 这个拍照软件基于 Python + OpenCV + Tkinter 实现，能满足 Windows 下基础的拍照、预览、保存需求，界面简洁易操作；
2. 核心逻辑是通过 OpenCV 调用摄像头获取实时画面，TKinter 搭建界面，点击按钮时保存当前帧为图片文件；
3. 支持打包成 EXE 可执行文件，无需 Python 环境也能直接运行，适配大多数 Windows 电脑。

你希望在这个Windows拍照软件里新增对焦功能，让拍摄的照片更清晰，对吧？我会在原有代码基础上增加手动对焦的按钮，通过调整摄像头的对焦参数实现这个功能，同时保持界面简洁、操作直观。

### 实现思路
1. 利用OpenCV的`set()`方法调整摄像头的对焦参数（`cv2.CAP_PROP_FOCUS`）；
2. 新增「对焦+」「对焦-」按钮，分别实现增加/降低对焦值，方便手动调节；
3. 增加对焦值显示，让你直观看到当前对焦参数；
4. 兼容不同摄像头（部分摄像头不支持手动对焦时，会给出友好提示）。

### 完整修改后代码
```python
import cv2
import tkinter as tk
from tkinter import messagebox
from PIL import Image, ImageTk
import datetime
import os

# 确保保存照片的文件夹存在
SAVE_DIR = "CameraPhotos"
if not os.path.exists(SAVE_DIR):
    os.makedirs(SAVE_DIR)

class WindowsCameraApp:
    def __init__(self, window):
        self.window = window
        self.window.title("Windows 简易拍照软件（带对焦功能）")
        self.window.geometry("800x700")  # 调整窗口大小适配新按钮
        self.window.resizable(False, False)  # 固定窗口大小

        # 初始化摄像头
        self.cap = cv2.VideoCapture(0)  # 0 表示默认摄像头，多个摄像头可换 1、2 等
        if not self.cap.isOpened():
            messagebox.showerror("错误", "无法打开摄像头！请检查摄像头是否被占用或驱动正常。")
            self.window.quit()
            return
        
        # 检查摄像头是否支持手动对焦
        self.focus_supported = self.check_focus_support()
        # 初始化对焦值（0-255 为常见对焦范围，不同摄像头范围可能不同）
        self.current_focus = 0
        if self.focus_supported:
            # 获取摄像头默认对焦值（无默认则设为100）
            self.current_focus = int(self.cap.get(cv2.CAP_PROP_FOCUS)) or 100
            self.cap.set(cv2.CAP_PROP_FOCUS, self.current_focus)

        # 创建预览标签
        self.label = tk.Label(window)
        self.label.pack(pady=10)

        # 创建按钮框架
        btn_frame = tk.Frame(window)
        btn_frame.pack(pady=10)

        # 对焦按钮组（仅当摄像头支持时显示）
        if self.focus_supported:
            # 对焦-按钮
            self.focus_minus_btn = tk.Button(
                btn_frame, text="对焦-", font=("微软雅黑", 12),
                width=8, command=lambda: self.adjust_focus(-10)
            )
            self.focus_minus_btn.grid(row=0, column=0, padx=5)

            # 对焦值显示标签
            self.focus_label = tk.Label(
                btn_frame, text=f"当前对焦值：{self.current_focus}",
                font=("微软雅黑", 12), width=15
            )
            self.focus_label.grid(row=0, column=1, padx=5)

            # 对焦+按钮
            self.focus_plus_btn = tk.Button(
                btn_frame, text="对焦+", font=("微软雅黑", 12),
                width=8, command=lambda: self.adjust_focus(10)
            )
            self.focus_plus_btn.grid(row=0, column=2, padx=5)

        # 拍照按钮
        self.capture_btn = tk.Button(
            btn_frame, text="拍照", font=("微软雅黑", 12),
            width=10, command=self.take_photo
        )
        self.capture_btn.grid(row=1, column=0, padx=10, pady=5)

        # 退出按钮
        self.quit_btn = tk.Button(
            btn_frame, text="退出", font=("微软雅黑", 12),
            width=10, command=self.close_app
        )
        self.quit_btn.grid(row=1, column=2, padx=10, pady=5)

        # 显示保存路径提示
        tip_label = tk.Label(
            window, text=f"照片将保存到：{os.path.abspath(SAVE_DIR)}",
            font=("微软雅黑", 10), fg="gray"
        )
        tip_label.pack(pady=5)

        # 对焦支持提示
        focus_tip = tk.Label(
            window, 
            text="提示：若未显示对焦按钮，说明当前摄像头不支持手动对焦",
            font=("微软雅黑", 9), fg="orange"
        )
        focus_tip.pack(pady=2)

        # 实时更新摄像头画面
        self.update_frame()

    def check_focus_support(self):
        """检查摄像头是否支持手动对焦"""
        try:
            # 尝试获取对焦参数范围，判断是否支持手动对焦
            min_focus = self.cap.get(cv2.CAP_PROP_FOCUS)
            # 部分摄像头返回 -1 表示不支持
            if min_focus == -1.0:
                return False
            return True
        except:
            return False

    def adjust_focus(self, step):
        """调整对焦值：step为正数增加对焦，负数降低对焦"""
        # 限定对焦值范围（0-255 是主流摄像头的对焦范围）
        new_focus = max(0, min(255, self.current_focus + step))
        # 设置新的对焦值
        self.cap.set(cv2.CAP_PROP_FOCUS, new_focus)
        # 更新当前对焦值并刷新显示
        self.current_focus = new_focus
        self.focus_label.config(text=f"当前对焦值：{self.current_focus}")

    def update_frame(self):
        """更新摄像头预览画面"""
        ret, frame = self.cap.read()
        if ret:
            # 转换颜色空间（OpenCV 是 BGR，PIL 是 RGB）
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            # 调整画面大小适配窗口
            frame = cv2.resize(frame, (760, 480))
            # 转换为 PIL 图像格式
            img = Image.fromarray(frame)
            imgtk = ImageTk.PhotoImage(image=img)
            self.label.imgtk = imgtk
            self.label.config(image=imgtk)
        # 循环更新画面（每10毫秒刷新一次）
        self.window.after(10, self.update_frame)

    def take_photo(self):
        """拍照并保存照片"""
        ret, frame = self.cap.read()
        if ret:
            # 生成带时间戳的文件名，避免重复
            timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
            photo_path = os.path.join(SAVE_DIR, f"photo_{timestamp}.jpg")
            # 保存照片
            cv2.imwrite(photo_path, frame)
            messagebox.showinfo("成功", f"照片已保存！\n路径：{photo_path}")
        else:
            messagebox.showerror("错误", "拍照失败，请重试！")

    def close_app(self):
        """关闭软件，释放摄像头"""
        self.cap.release()  # 释放摄像头资源
        self.window.quit()  # 关闭窗口

if __name__ == "__main__":
    root = tk.Tk()
    app = WindowsCameraApp(root)
    root.mainloop()
```

### 新增功能说明
1. **对焦功能核心逻辑**：
   - `check_focus_support()`：启动时自动检测摄像头是否支持手动对焦，若不支持则不显示对焦按钮，避免无效操作；
   - `adjust_focus(step)`：点击「对焦+」「对焦-」时调用，每次调整步长为10（可修改代码中`step`参数调整灵敏度），同时限定对焦值在0-255（主流摄像头的通用范围）；
   - 新增「当前对焦值」标签，实时显示参数，方便精准调节。

2. **使用步骤（新增部分）**：
   - 运行软件后，若摄像头支持手动对焦，会显示「对焦-」「当前对焦值」「对焦+」按钮；
   - 预览画面时，点击「对焦+」增加对焦值（画面会更清晰对准近处物体），点击「对焦-」降低对焦值（适合拍摄远处物体）；
   - 调节到画面最清晰的对焦值后，点击「拍照」即可保存清晰的照片。

3. **注意事项**：
   - 部分笔记本内置摄像头、廉价USB摄像头可能不支持手动对焦（此时不会显示对焦按钮，软件会给出提示）；
   - 不同摄像头的对焦值范围可能略有差异，若0-255范围不合适，可修改`adjust_focus`方法中的`max(0, min(255, ...))`数值；
   - 调节对焦时建议缓慢点击按钮，观察预览画面的清晰度变化，找到最佳值。

### 总结
1. 新增的对焦功能通过OpenCV的`CAP_PROP_FOCUS`参数实现手动调节，适配多数支持对焦的Windows摄像头；
2. 软件会自动检测摄像头对焦支持性，无适配性问题时隐藏对焦按钮，避免操作困惑；
3. 对焦调节步长设为10，范围限定0-255，兼顾调节精度和易用性，可根据需求自行修改。
