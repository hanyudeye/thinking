你想要用网页技术（HTML+CSS+JavaScript）实现一个适配手机横竖屏的雷霆战机小游戏，包含摇杆操控、双发射口敌机、大招清屏、5分钟刷新BOSS、两关BOSS手动切换等功能，这个需求完全可以实现。

下面给你一份**完整的可直接运行代码**，基于原生JS+Canvas实现，无需依赖任何框架，手机端可直接打开，满足你所有核心需求：

```html
<body>
    <!-- 游戏画布 -->
    <canvas id="gameCanvas"></canvas>

    <!-- 虚拟摇杆 -->
    <div class="joystick" id="joystick">
        <div class="knob" id="joystickKnob"></div>
    </div>

    <!-- 大招按钮 -->
    <button class="skill-btn" id="skillBtn">大招清屏</button>

    <!-- 下一个BOSS按钮 -->
    <button class="next-boss-btn" id="nextBossBtn">下一个BOSS →</button>

    <!-- 游戏信息 -->
    <div class="game-info" id="gameInfo">
        关卡：1 | BOSS剩余时间：5:00 | 状态：战斗中
    </div>

    <script>
        // ************************ 全局配置 ************************
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const joystick = document.getElementById('joystick');
        const joystickKnob = document.getElementById('joystickKnob');
        const skillBtn = document.getElementById('skillBtn');
        const nextBossBtn = document.getElementById('nextBossBtn');
        const gameInfo = document.getElementById('gameInfo');

        // 游戏状态
        let gameState = {
            width: window.innerWidth,
            height: window.innerHeight,
            isMobile: window.innerWidth <= 768,
            isSkillActive: false, // 大招是否激活
            bossLevel: 1, // 当前BOSS关卡
            bossTimer: 5 * 60, // BOSS倒计时（秒）
            bossIsDead: false, // BOSS是否被击败
            entities: { // 游戏实体
                player: null,
                boss: null,
                playerBullets: [],
                enemyBullets: [],
                explosions: []
            },
            joystick: { // 摇杆状态
                isPressed: false,
                baseX: 0,
                baseY: 0,
                offsetX: 0,
                offsetY: 0,
                maxDistance: 60
            }
        };

        // ************************ 工具类 ************************
        // 碰撞检测（矩形碰撞）
        function isColliding(a, b) {
            return a.x < b.x + b.width &&
                   a.x + a.width > b.x &&
                   a.y < b.y + b.height &&
                   a.y + a.height > b.y;
        }

        // 格式化时间（秒 → 分:秒）
        function formatTime(seconds) {
            const min = Math.floor(seconds / 60);
            const sec = seconds % 60;
            return `${min}:${sec.toString().padStart(2, '0')}`;
        }

        // 随机数生成
        function random(min, max) {
            return Math.random() * (max - min) + min;
        }

        // ************************ 游戏实体类 ************************
        // 玩家飞机类
        class Player {
            constructor() {
                this.width = 80;
                this.height = 80;
                this.x = gameState.width / 2 - this.width / 2;
                this.y = gameState.height - this.height - 120;
                this.speed = 5;
                this.hp = 1;
                this.color = '#00ffdd';
                this.fireRate = 200; // 子弹发射间隔（毫秒）
                this.lastFireTime = 0;
            }

            // 更新玩家位置（根据摇杆）
            update() {
                // 摇杆控制移动
                if (gameState.joystick.isPressed) {
                    const moveX = gameState.joystick.offsetX * 0.1;
                    const moveY = gameState.joystick.offsetY * 0.1;

                    // 边界检测
                    this.x = Math.max(0, Math.min(gameState.width - this.width, this.x + moveX));
                    this.y = Math.max(0, Math.min(gameState.height - this.height, this.y + moveY));
                }
            }

            // 绘制玩家飞机
            draw() {
                ctx.save();
                // 绘制飞机主体（简化版）
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.moveTo(this.x + this.width / 2, this.y);
                ctx.lineTo(this.x, this.y + this.height);
                ctx.lineTo(this.x + this.width, this.y + this.height);
                ctx.closePath();
                ctx.fill();

                // 绘制飞机细节
                ctx.strokeStyle = '#fff';
                ctx.lineWidth = 2;
                ctx.stroke();
                ctx.restore();
            }

            // 发射子弹
            fire(timestamp) {
                if (timestamp - this.lastFireTime > this.fireRate) {
                    // 玩家单发射口（居中）
                    const bulletX = this.x + this.width / 2 - 5;
                    const bulletY = this.y - 10;

                    gameState.entities.playerBullets.push(new PlayerBullet(bulletX, bulletY));
                    this.lastFireTime = timestamp;
                }
            }
        }

        // BOSS类（两关不同属性）
        class Boss {
            constructor(level) {
                this.level = level;
                this.width = level === 1 ? 200 : 250;
                this.height = level === 1 ? 150 : 180;
                this.x = gameState.width / 2 - this.width / 2;
                this.y = 50;
                this.hp = level === 1 ? 100 : 150; // 第二关血量更高
                this.speed = level === 1 ? 1 : 1.5; // 第二关移动更快
                this.color = level === 1 ? '#ff4444' : '#ff8800';
                this.fireRate = level === 1 ? 1000 : 800; // 第二关发射更快
                this.lastFireTime = 0;
                this.firePoints = [ // 两个发射口
                    { x: this.x + 30, y: this.y + this.height - 20 },
                    { x: this.x + this.width - 30, y: this.y + this.height - 20 }
                ];
                this.moveDirection = 1; // 左右移动方向
            }

            // 更新BOSS状态（移动+发射子弹）
            update(timestamp) {
                if (gameState.bossIsDead) return;

                // 左右往返移动
                this.x += this.speed * this.moveDirection;
                if (this.x <= 0 || this.x + this.width >= gameState.width) {
                    this.moveDirection *= -1;
                }

                // 更新发射口位置
                this.firePoints = [
                    { x: this.x + 30, y: this.y + this.height - 20 },
                    { x: this.x + this.width - 30, y: this.y + this.height - 20 }
                ];

                // 发射子弹（两个发射口发射不同类型子弹）
                if (timestamp - this.lastFireTime > this.fireRate) {
                    this.firePoints.forEach((point, index) => {
                        if (this.level === 1) {
                            // 第一关：普通直线子弹（两个发射口同类型）
                            gameState.entities.enemyBullets.push(new EnemyBullet(point.x, point.y, index));
                        } else {
                            // 第二关：发射口1直线子弹，发射口2跟踪子弹（简单跟踪）
                            if (index === 0) {
                                gameState.entities.enemyBullets.push(new EnemyBullet(point.x, point.y, 0));
                            } else {
                                gameState.entities.enemyBullets.push(new EnemyBullet(point.x, point.y, 1, true));
                            }
                        }
                    });
                    this.lastFireTime = timestamp;
                }
            }

            // 绘制BOSS
            draw() {
                if (gameState.bossIsDead) return;

                ctx.save();
                // 绘制BOSS主体
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.width, this.height);

                // 绘制两个发射口
                ctx.fillStyle = '#000';
                this.firePoints.forEach(point => {
                    ctx.beginPath();
                    ctx.arc(point.x, point.y, 15, 0, Math.PI * 2);
                    ctx.fill();
                });

                // 绘制血量条
                ctx.fillStyle = '#ff0000';
                ctx.fillRect(this.x, this.y - 20, this.width, 10);
                ctx.fillStyle = '#00ff00';
                ctx.fillRect(this.x, this.y - 20, (this.hp / (this.level === 1 ? 100 : 150)) * this.width, 10);

                // 绘制关卡标识
                ctx.fillStyle = '#fff';
                ctx.font = '20px Arial';
                ctx.fillText(`BOSS ${this.level}`, this.x + this.width / 2 - 30, this.y + this.height / 2);
                ctx.restore();
            }

            // 受到伤害
            takeDamage(damage = 1) {
                this.hp -= damage;
                if (this.hp <= 0) {
                    gameState.bossIsDead = true;
                    gameState.bossTimer = 0;
                    nextBossBtn.style.display = 'block';
                    this.createExplosion();
                }
            }

            // 击败时生成爆炸效果
            createExplosion() {
                for (let i = 0; i < 20; i++) {
                    gameState.entities.explosions.push(new Explosion(
                        this.x + random(0, this.width),
                        this.y + random(0, this.height)
                    ));
                }
            }
        }

        // 玩家子弹类
        class PlayerBullet {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.width = 10;
                this.height = 20;
                this.speed = 10;
                this.color = '#00ffdd';
            }

            update() {
                this.y -= this.speed;
            }

            draw() {
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.width, this.height);
            }

            // 是否超出屏幕
            isOffScreen() {
                return this.y + this.height < 0;
            }
        }

        // 敌机子弹类
        class EnemyBullet {
            constructor(x, y, type, isTracking = false) {
                this.x = x;
                this.y = y;
                this.width = 10;
                this.height = 20;
                this.speed = type === 0 ? 5 : 7;
                this.color = type === 0 ? '#ff4444' : '#ffff00';
                this.isTracking = isTracking; // 是否跟踪玩家
                this.trackingSpeed = 0.5;
            }

            update() {
                // 跟踪玩家逻辑
                if (this.isTracking && gameState.entities.player) {
                    const player = gameState.entities.player;
                    const dx = player.x + player.width / 2 - (this.x + this.width / 2);
                    const dy = player.y + player.height / 2 - (this.y + this.height / 2);
                    const distance = Math.sqrt(dx * dx + dy * dy);

                    if (distance > 0) {
                        this.x += (dx / distance) * this.trackingSpeed;
                        this.y += (dy / distance) * this.trackingSpeed;
                    }
                } else {
                    this.y += this.speed;
                }
            }

            draw() {
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.width, this.height);
            }

            // 是否超出屏幕
            isOffScreen() {
                return this.y > gameState.height;
            }
        }

        // 爆炸效果类
        class Explosion {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.size = random(10, 30);
                this.alpha = 1;
                this.speed = 0.05;
                this.color = `rgb(${random(200, 255)}, ${random(0, 100)}, ${random(0, 50)})`;
            }

            update() {
                this.alpha -= this.speed;
                this.size += 0.5;
            }

            draw() {
                ctx.save();
                ctx.globalAlpha = this.alpha;
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }

            // 是否消失
            isGone() {
                return this.alpha <= 0;
            }
        }

        // ************************ 游戏逻辑 ************************
        // 初始化游戏
        function initGame() {
            // 适配画布大小
            resizeCanvas();

            // 初始化玩家和BOSS
            gameState.entities.player = new Player();
            gameState.entities.boss = new Boss(gameState.bossLevel);

            // 重置游戏状态
            gameState.bossIsDead = false;
            gameState.bossTimer = 5 * 60;
            nextBossBtn.style.display = 'none';
            updateGameInfo();

            // 开始游戏循环
            requestAnimationFrame(gameLoop);
        }

        // 调整画布大小（适配横竖屏）
        function resizeCanvas() {
            gameState.width = window.innerWidth;
            gameState.height = window.innerHeight;
            gameState.isMobile = window.innerWidth <= 768;

            canvas.width = gameState.width;
            canvas.height = gameState.height;

            // 重置摇杆基础位置
            gameState.joystick.baseX = joystick.offsetLeft + joystick.offsetWidth / 2;
            gameState.joystick.baseY = joystick.offsetTop + joystick.offsetHeight / 2;
        }

        // 更新游戏信息面板
        function updateGameInfo() {
            const status = gameState.bossIsDead ? 'BOSS已击败' : '战斗中';
            gameInfo.innerHTML = `
                关卡：${gameState.bossLevel} | 
                BOSS剩余时间：${formatTime(gameState.bossTimer)} | 
                状态：${status}
            `;
        }

        // 大招清屏功能
        function activateSkill() {
            if (gameState.isSkillActive || gameState.bossIsDead) return;

            gameState.isSkillActive = true;
            // 清除所有敌机子弹
            gameState.entities.enemyBullets = [];

            // 大招视觉效果（简单闪烁）
            ctx.fillStyle = 'rgba(255, 255, 255, 0.5)';
            ctx.fillRect(0, 0, gameState.width, gameState.height);

            // 3秒后关闭大招状态
            setTimeout(() => {
                gameState.isSkillActive = false;
            }, 3000);
        }

        // 切换到下一个BOSS
        function nextBoss() {
            if (gameState.bossLevel >= 2) {
                alert('恭喜！已通关所有BOSS关卡！');
                gameState.bossLevel = 1;
            } else {
                gameState.bossLevel++;
            }

            // 重置实体和状态
            gameState.entities.playerBullets = [];
            gameState.entities.enemyBullets = [];
            gameState.entities.explosions = [];
            gameState.entities.boss = new Boss(gameState.bossLevel);

            // 重置计时器和按钮
            gameState.bossIsDead = false;
            gameState.bossTimer = 5 * 60;
            nextBossBtn.style.display = 'none';
            updateGameInfo();
        }

        // 游戏主循环
        function gameLoop(timestamp) {
            // 清屏
            ctx.clearRect(0, 0, gameState.width, gameState.height);

            // 倒计时更新（BOSS未被击败时）
            if (!gameState.bossIsDead) {
                // 每秒钟更新一次倒计时
                if (Math.floor(timestamp / 1000) !== Math.floor((timestamp - 16) / 1000)) {
                    gameState.bossTimer = Math.max(0, gameState.bossTimer - 1);
                    updateGameInfo();

                    // 倒计时结束，BOSS未被击败（玩家失败）
                    if (gameState.bossTimer === 0) {
                        alert(`时间到！BOSS ${gameState.bossLevel} 未被击败，游戏失败！`);
                        initGame();
                        return;
                    }
                }
            }

            // 更新并绘制玩家
            const player = gameState.entities.player;
            if (player) {
                player.update();
                player.draw();
                player.fire(timestamp); // 自动发射子弹
            }

            // 更新并绘制BOSS
            const boss = gameState.entities.boss;
            if (boss) {
                boss.update(timestamp);
                boss.draw();
            }

            // 更新并绘制玩家子弹（含碰撞检测）
            gameState.entities.playerBullets = gameState.entities.playerBullets.filter(bullet => {
                bullet.update();
                bullet.draw();

                // 子弹与BOSS碰撞检测
                if (!gameState.bossIsDead && boss && isColliding(bullet, boss)) {
                    boss.takeDamage();
                    return false;
                }

                return !bullet.isOffScreen();
            });

            // 更新并绘制敌机子弹（含碰撞检测）
            gameState.entities.enemyBullets = gameState.entities.enemyBullets.filter(bullet => {
                bullet.update();
                bullet.draw();

                // 子弹与玩家碰撞检测（玩家未无敌）
                if (!gameState.isSkillActive && player && isColliding(bullet, player)) {
                    // 玩家被击中，游戏结束
                    alert('你被击中了！游戏失败！');
                    initGame();
                    return false;
                }

                return !bullet.isOffScreen();
            });

            // 更新并绘制爆炸效果
            gameState.entities.explosions = gameState.entities.explosions.filter(explosion => {
                explosion.update();
                explosion.draw();
                return !explosion.isGone();
            });

            // 继续游戏循环
            requestAnimationFrame(gameLoop);
        }

        // ************************ 事件监听 ************************
        // 窗口大小变化（适配横竖屏切换）
        window.addEventListener('resize', resizeCanvas);

        // 摇杆事件（移动端触摸）
        joystickKnob.addEventListener('touchstart', (e) => {
            e.preventDefault();
            gameState.joystick.isPressed = true;
            const touch = e.touches[0];
            gameState.joystick.baseX = touch.clientX;
            gameState.joystick.baseY = touch.clientY;
        });

        document.addEventListener('touchmove', (e) => {
            if (!gameState.joystick.isPressed) return;
            e.preventDefault();

            const touch = e.touches[0];
            const dx = touch.clientX - gameState.joystick.baseX;
            const dy = touch.clientY - gameState.joystick.baseY;
            const distance = Math.sqrt(dx * dx + dy * dy);

            // 限制摇杆最大距离
            if (distance > gameState.joystick.maxDistance) {
                const ratio = gameState.joystick.maxDistance / distance;
                gameState.joystick.offsetX = dx * ratio;
                gameState.joystick.offsetY = dy * ratio;
            } else {
                gameState.joystick.offsetX = dx;
                gameState.joystick.offsetY = dy;
            }

            // 更新摇杆视觉位置
            joystickKnob.style.transform = `translate(${gameState.joystick.offsetX}px, ${gameState.joystick.offsetY}px)`;
        });

        document.addEventListener('touchend', () => {
            if (!gameState.joystick.isPressed) return;

            gameState.joystick.isPressed = false;
            gameState.joystick.offsetX = 0;
            gameState.joystick.offsetY = 0;

            // 重置摇杆位置
            joystickKnob.style.transform = 'translate(-50%, -50%)';
        });

        // 大招按钮点击/触摸事件
        skillBtn.addEventListener('click', activateSkill);
        skillBtn.addEventListener('touchstart', (e) => {
            e.preventDefault();
            activateSkill();
        });

        // 下一个BOSS按钮事件
        nextBossBtn.addEventListener('click', nextBoss);
        nextBossBtn.addEventListener('touchstart', (e) => {
            e.preventDefault();
            nextBoss();
        });

        // ************************ 启动游戏 ************************
        window.onload = initGame;
    </script>
</body>
</html>
```

## 一、 功能说明（完全满足你的需求）
1.  **横竖屏适配**：画布自动跟随窗口大小调整，手机横屏/竖屏切换均可正常运行
2.  **双发射口敌机**：
    - BOSS有两个固定发射口，位于机身下方左右两侧
    - 第1关：双发射口发射红色直线子弹
    - 第2关：左发射口红色直线子弹，右发射口黄色跟踪子弹（难度提升）
3.  **虚拟摇杆操控**：手机端左侧下方显示摇杆，可拖动控制飞机在屏幕内任意移动（有边界限制，不会飞出屏幕）
4.  **大招清屏**：右侧下方红色圆形按钮，点击后立即清除所有敌机子弹
5.  **BOSS机制**：
    - 初始为第1关BOSS，5分钟倒计时（秒级更新）
    - 击败BOSS后弹出「下一个BOSS」按钮，手动点击切换到第2关
    - 第2关BOSS血量更高、移动更快、发射频率更高，还有跟踪子弹
    - 玩家飞机触碰任意敌机子弹立即游戏失败，倒计时结束未击败BOSS也失败
6.  **视觉反馈**：BOSS被击败时有爆炸效果，有血量条显示BOSS剩余血量，游戏信息面板显示关卡和倒计时
